using namespace hip::wmma;

__global__ void wmma_gemm_kernel(
    half* A, half* B, float* C,
    int M, int N, int K)
{
    const int WMMA_M = 16, WMMA_N = 16, WMMA_K = 16;

    int warpM = (blockIdx.x * blockDim.x + threadIdx.x) / warpSize;
    int warpN = (blockIdx.y * blockDim.y + threadIdx.y);

    fragment<matrix_a, WMMA_M, WMMA_N, WMMA_K, half, row_major> a_frag;
    fragment<matrix_b, WMMA_M, WMMA_N, WMMA_K, half, col_major> b_frag;
    fragment<accumulator, WMMA_M, WMMA_N, WMMA_K, float> acc_frag;

    fill_fragment(acc_frag, 0.0f);

    for (int i = 0; i < K; i += WMMA_K) {
        int aRow = warpM * WMMA_M, aCol = i;
        int bRow = i,              bCol = warpN * WMMA_N;

        if (aRow < M && aCol < K && bRow < K && bCol < N) {
            load_matrix_sync(a_frag, A + aRow * K + aCol, K);
            load_matrix_sync(b_frag, B + bRow * N + bCol, N);
            mma_sync(acc_frag, a_frag, b_frag, acc_frag);
        }
    }

    int cRow = warpM * WMMA_M, cCol = warpN * WMMA_N;
    if (cRow < M && cCol < N)
        store_matrix_sync(C + cRow * N + cCol, acc_frag, N, mem_row_major);
}

__global__ void grid_sync_reduction(float* data, float* result, int N) {
    hip::grid_group grid = hip::this_grid();

    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    __shared__ float sdata[256];

    sdata[threadIdx.x] = (idx < N) ? data[idx] : 0.0f;
    __syncthreads();

    // Local block reduction
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (threadIdx.x < s) sdata[threadIdx.x] += sdata[threadIdx.x + s];
        __syncthreads();
    }

    if (threadIdx.x == 0) result[blockIdx.x] = sdata[0];

    // GRID-LEVEL SYNC: all blocks must be alive simultaneously
    grid.sync();

    // Only thread 0 of block 0 does final reduction
    if (blockIdx.x == 0 && threadIdx.x == 0) {
        float total = 0.0f;
        for (int i = 0; i < gridDim.x; i++) total += result[i];
        result[0] = total;
    }
}

__global__ void ptx_ldmatrix_kernel(half* src, half* dst, int stride) {
    __shared__ half smem[16 * 16];

    int tid = threadIdx.x;
    int lane = tid % 64;

    // ROCm does not have a direct 1:1 equivalent to cp.async or ldmatrix.
    // These are NVIDIA-specific PTX instructions for Tensor Core memory movement.
    // The native HIP equivalent for moving data to shared memory and loading into registers:
    
    half* smem_ptr = &smem[lane * 16];
    half* gmem_ptr = src + lane * stride;

    // Global to Shared load
    for (int i = 0; i < 16; ++i) {
        smem_ptr[i] = gmem_ptr[i];
    }
    
    __syncthreads();

    // Load from shared to registers (simulating ldmatrix 8x8x2)
    // In ROCm, this is typically handled by vector loads (buffer_load or standard loads)
    uint32_t ra, rb;
    
    // Load two 32-bit words (4 halfs each) to simulate the register output
    ra = *reinterpret_cast<uint32_t*>(&smem_ptr[0]);
    rb = *reinterpret_cast<uint32_t*>(&smem_ptr[4]);

    // Store results back
    reinterpret_cast<uint32_t*>(dst)[lane * 2 + 0] = ra;
    reinterpret_cast<uint32_t*>(dst)[lane * 2 + 1] = rb;
}

```cpp
#include <hip/hip_runtime.h>

__global__ void persistent_producer_consumer(
    volatile int* work_queue,
    volatile int* done_flags,
    float* data,
    int num_tasks,
    int spin_cycles)
{
    int tid = hipBlockIdx_x * hipBlockDim_x + hipThreadIdx_x;

    while (true) {
        // Spin-poll for a task slot (timed spin using GPU clock)
        int task = -1;
        long long start = clock64();

        while (task < 0) {
            // Try to atomically grab a task
            for (int i = 0; i < num_tasks; i++) {
                if (atomicCAS((int*)&done_flags[i], 0, 1) == 0) {
                    task = i;
                    break;
                }
            }
            // Bail if we've spun for too long (clock-based timeout)
            if ((clock64() - start) > spin_cycles) break;
        }

        if (task < 0) break;

        // Do the work
        data[task * hipBlockDim_x + hipThreadIdx_x] *= 1.5f;
        __threadfence();
        done_flags[task] = 2;  // Mark complete
    }
}

#include <hip/hip_runtime.h>

__global__ void nanosleep_sync_kernel(
    volatile int* lock,
    float* shared_buffer,
    float* output,
    int N)
{
    int tid = blockIdx.x * blockDim.x + threadIdx.x;

    // Spin-lock with nanosleep backoff (reduces memory traffic)
    if (threadIdx.x == 0) {
        int backoff = 8;
        while (atomicCAS((int*)lock, 0, 1) != 0) {
            __nanosleep(backoff);  // ROCm/HIP compatible: sleep this thread N nanoseconds
            backoff = min(backoff * 2, 256);  // Exponential backoff
        }
    }
    __syncthreads();

    // Critical section -- modify shared buffer
    if (tid < N) shared_buffer[threadIdx.x] += 1.0f;

    // Acquire/release fence to ensure ordering
    __threadfence();  // Full device-level memory fence

    __syncthreads();
    if (threadIdx.x == 0) {
        atomicExch((int*)lock, 0);  // Release lock
    }

    if (tid < N) output[tid] = shared_buffer[threadIdx.x];
}

// CUTLASS-inspired warp-specialized GEMM with double-buffered shared memory
#pragma once
#include <hip/hip_runtime.h>
#include <hip/hip_fp16.h>

template <int BM, int BN, int BK, int WM, int WN, int WK>
struct GemmPolicy {
    static constexpr int kBlockM = BM, kBlockN = BN, kBlockK = BK;
    static constexpr int kWmmaM = WM, kWmmaN = WN, kWmmaK = WK;
    static constexpr int kWarpsM = BM / WM, kWarpsN = BN / WN;
};

template <typename P>
struct SharedTile {
    half A[P::kBlockM][P::kBlockK];
    half B[P::kBlockK][P::kBlockN];
};

template <typename Policy>
__global__ void __launch_bounds__(256)
cutlass_gemm_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B,
    float*        __restrict__ C,
    int M, int N, int K)
{
    using P = Policy;
    __shared__ SharedTile<P> smem[2];
    const int warp_id = threadIdx.x / 64; // ROCm wave size is typically 64
    const int warp_m  = warp_id / P::kWarpsN;
    const int warp_n  = warp_id % P::kWarpsN;
    
    float acc_frag[P::kWmmaM * P::kWmmaN] = {0.0f};
    int smem_idx = 0;

    for (int k = 0; k < K; k += P::kBlockK) {
        for (int i = threadIdx.x; i < P::kBlockM * P::kBlockK; i += blockDim.x) {
            int row = i / P::kBlockK, col = i % P::kBlockK;
            int ga = (blockIdx.x * P::kBlockM + row) * K + (k + col);
            smem[smem_idx].A[row][col] = (blockIdx.x * P::kBlockM + row < M && k + col < K)
                ? A[ga] : __float2half(0.0f);
        }
        for (int i = threadIdx.x; i < P::kBlockK * P::kBlockN; i += blockDim.x) {
            int row = i / P::kBlockN, col = i % P::kBlockN;
            int gb = (k + row) * N + (blockIdx.y * P::kBlockN + col);
            smem[smem_idx].B[row][col] = (k + row < K && blockIdx.y * P::kBlockN + col < N)
                ? B[gb] : __float2half(0.0f);
        }
        __syncthreads();

        for (int kk = 0; kk < P::kBlockK; kk += P::kWmmaK) {
            // HIP does not have a 1:1 header-only equivalent to nvcuda::wmma.
            // This section implements the logical equivalent of the MMA operation.
            for (int m = 0; m < P::kWmmaM; ++m) {
                for (int n = 0; n < P::kWmmaN; ++n) {
                    float sum = 0.0f;
                    for (int k_inner = 0; k_inner < P::kWmmaK; ++k_inner) {
                        sum += __half2float(smem[smem_idx].A[warp_m * P::kWmmaM + m][kk + k_inner]) * 
                               __half2float(smem[smem_idx].B[kk + k_inner][warp_n * P::kWmmaN + n]);
                    }
                    acc_frag[m * P::kWmmaN + n] += sum;
                }
            }
        }
        smem_idx ^= 1;
        __syncthreads();
    }

    int global_m = blockIdx.x * P::kBlockM + warp_m * P::kWmmaM;
    int global_n = blockIdx.y * P::kBlockN + warp_n * P::kWmmaN;
    
    if (global_m < M && global_n < N) {
        for (int m = 0; m < P::kWmmaM && (global_m + m) < M; ++m) {
            for (int n = 0; n < P::kWmmaN && (global_n + n) < N; ++n) {
                C[(global_m + m) * N + (global_n + n)] = acc_frag[m * P::kWmmaN + n];
            }
        }
    }
}

using DefaultPolicy = GemmPolicy<64, 64, 16, 16, 16, 16>;
void launch_cutlass_gemm(half* A, half* B, float* C, int M, int N, int K) {
    dim3 grid((M + 63) / 64, (N + 63) / 64);
    hipLaunchKernelGGL(cutlass_gemm_kernel, grid, 256, 0, 0, A, B, C, M, N, K);
}

#pragma once
#include <hip/hip_runtime.h>

template <int BYTES>
__device__ __forceinline__ void async_copy(void* dst_smem, const void* src_gmem) {
    // ROCm uses buffer loads/stores or standard copies; 
    // there is no direct 1:1 equivalent to NVIDIA's cp.async.
    // For native HIP, we use a standard copy.
    if (BYTES == sizeof(half)) {
        *(half*)dst_smem = *(const half*)src_gmem;
    } else {
        memcpy(dst_smem, src_gmem, BYTES);
    }
}

__device__ __forceinline__ void async_commit() {
    // No-op: HIP/AMD memory model handles consistency differently
}

template <int N_PENDING>
__device__ __forceinline__ void async_wait() {
    // No-op: HIP/AMD memory model handles consistency differently
}

// FILE: pipeline_gemm.cpp
#include <hip/hip_runtime.h>
#include <hip/hip_fp16.h>
#include <hip/hip_mm.h>

constexpr int PIPE_STAGES = 3;
constexpr int TILE_M = 64, TILE_N = 64, TILE_K = 16;

struct alignas(128) SmemBuf {
    half A[PIPE_STAGES][TILE_M][TILE_K];
    half B[PIPE_STAGES][TILE_K][TILE_N];
};

__global__ void pipeline_gemm(
    const half* __restrict__ A, const half* __restrict__ B,
    float* __restrict__ C, int M, int N, int K)
{
    extern __shared__ SmemBuf smem_buf;
    const int warp_id = threadIdx.x / 32;
    
    hipmm::fragment<hipmm::accumulator, 16, 16, 16, float> acc;
    hipmm::fill_fragment(acc, 0.0f);
    
    int num_tiles = (K + TILE_K - 1) / TILE_K;

    // Prologue: prefill pipeline
    for (int s = 0; s < (PIPE_STAGES - 1 < num_tiles ? PIPE_STAGES - 1 : num_tiles); ++s) {
        int k_off = s * TILE_K;
        for (int i = threadIdx.x; i < TILE_M * TILE_K; i += blockDim.x)
            async_copy<sizeof(half)>(
                &smem_buf.A[s][i/TILE_K][i%TILE_K],
                A + (blockIdx.x * TILE_M + i/TILE_K) * K + k_off + i%TILE_K);
        for (int i = threadIdx.x; i < TILE_K * TILE_N; i += blockDim.x)
            async_copy<sizeof(half)>(
                &smem_buf.B[s][i/TILE_N][i%TILE_N],
                B + (k_off + i/TILE_N) * N + blockIdx.y * TILE_N + i%TILE_N);
        async_commit();
    }

    // Steady-state: overlap compute with prefetch
    for (int tile = 0; tile < num_tiles; ++tile) {
        int prefetch = tile + PIPE_STAGES - 1;
        int cs = tile % PIPE_STAGES;
        
        async_wait<PIPE_STAGES - 2>();
        __syncthreads();

        if (prefetch < num_tiles) {
            int k_off = prefetch * TILE_K, ps = prefetch % PIPE_STAGES;
            for (int i = threadIdx.x; i < TILE_M * TILE_K; i += blockDim.x)
                async_copy<sizeof(half)>(
                    &smem_buf.A[ps][i/TILE_K][i%TILE_K],
                    A + (blockIdx.x * TILE_M + i/TILE_K) * K + k_off + i%TILE_K);
            for (int i = threadIdx.x; i < TILE_K * TILE_N; i += blockDim.x)
                async_copy<sizeof(half)>(
                    &smem_buf.B[ps][i/TILE_N][i%TILE_N],
                    B + (k_off + i/TILE_N) * N + blockIdx.y * TILE_N + i%TILE_N);
            async_commit();
        }

        hipmm::fragment<hipmm::matrix_a, 16, 16, 16, half, hipmm::row_major> a_frag;
        hipmm::fragment<hipmm::matrix_b, 16, 16, 16, half, hipmm::col_major> b_frag;
        
        hipmm::load_matrix_sync(a_frag, &smem_buf.A[cs][warp_id * 16][0], TILE_K);
        hipmm::load_matrix_sync(b_frag, &smem_buf.B[cs][0][warp_id * 16], TILE_N);
        hipmm::mma_sync(acc, a_frag, b_frag, acc);
    }

    async_wait<0>();
    __syncthreads();

    int out_m = blockIdx.x * TILE_M + warp_id * 16, out_n = blockIdx.y * TILE_N;
    if (out_m < M && out_n < N)
        hipmm::store_matrix_sync(C + out_m * N + out_n, acc, N, hipmm::mem_row_major);
}

// Warp-specialized: producer warps load, consumer warps compute.
// Using HIP's __syncthreads() as ROCm does not have a direct 1:1 equivalent 
// to PTX named barriers for arbitrary warp subsets without using 
// specific hardware-level memory fences or atomic-based barriers.
#include <hip/hip_runtime.h>

__device__ __forceinline__ void bar_arrive(int id, int n) {
    // ROCm does not support PTX bar.arrive. 
    // For a direct conversion of the logic, we use a memory fence.
    __threadfence_block();
}

__device__ __forceinline__ void bar_wait(int id, int n) {
    // ROCm does not support PTX bar.sync.
    // We use __syncthreads() to ensure all producer warps have completed.
    __syncthreads();
}

constexpr int PROD_WARPS = 2, CONS_WARPS = 6, TOT_WARPS = 8;
constexpr int SR = 64, SC = 64;

__global__ void
warp_specialized_stencil(const float* src, float* dst, int N)
{
    __shared__ float smem[SR][SC];
    const int warp_id = threadIdx.x / 64, lane = threadIdx.x % 64; // ROCm wavefront is 64
    
    if (warp_id < PROD_WARPS) {
        int rw = SR / PROD_WARPS, rs = warp_id * rw;
        for (int r = rs; r < rs + rw; ++r)
            for (int c = lane; c < SC; c += 64) {
                int gi = blockIdx.x * SR * SC + r * SC + c;
                smem[r][c] = gi < N ? src[gi] : 0.0f;
            }
        bar_arrive(0, TOT_WARPS * 64);
    } 
    
    // In HIP, to synchronize between different warps in the same block, 
    // __syncthreads() is the standard mechanism.
    __syncthreads();

    if (warp_id >= PROD_WARPS) {
        int cid = warp_id - PROD_WARPS;
        int rw = SR / CONS_WARPS, rs = cid * rw;
        for (int r = rs + 1; r < rs + rw - 1; ++r)
            for (int c = lane + 1; c < SC - 1; c += 64) {
                float v = 0.25f * (smem[r-1][c] + smem[r+1][c] +
                                   smem[r][c-1] + smem[r][c+1]);
                int go = blockIdx.x * SR * SC + r * SC + c;
                if (go < N) dst[go] = v;
            }
    }
}

__global__ void hip_ldmatrix_kernel(half* src, half* dst, int stride) {
    // Shared memory allocation: 16x16 halfs
    __shared__ half smem[16 * 16];

    int tid = threadIdx.x;
    int lane = tid % 32;

    // 1. Replace cp.async: Global to Shared Memory
    // We load 32 bytes (16 halfs) per thread to match the original PTX logic
    // src + lane * stride points to the start of the row
    for (int i = 0; i < 16; ++i) {
        smem[lane * 16 + i] = src[lane * stride + i];
    }

    // Ensure all threads have completed the copy to shared memory
    __syncthreads();

    // 2. Replace ldmatrix: Shared Memory to Registers
    // ldmatrix.x2 loads two 8x8 fragments. 
    // In the original PTX, this reshapes the 16x16 tile.
    // We emulate this by reading the specific 128-bit chunks.
    
    uint32_t ra = 0;
    uint32_t rb = 0;

    // It loads data such that registers are populated for MFMA/WMMA.
    // We cast to uint32_t to move 2 halfs at a time.
    
    // Fragment A (First 8x8 block)
    half* ptr_a = &smem[lane * 16]; 
    ra = *reinterpret_cast<uint32_t*>(ptr_a);

    half* ptr_b = &smem[lane * 16 + 8];
    rb = *reinterpret_cast<uint32_t*>(ptr_b);

    // Using uint32_t pointer to store the register pairs

/**
 * HIP implementation of a persistent producer-consumer kernel.
 * 
 * Reasoning for changes:
 * 1. __clock64() -> __clock(): AMD GPUs use __clock() for cycle counting.
 * 2. Memory fences: __threadfence() is natively supported in HIP for global visibility.
 * 3. Atomics: atomicCAS is a direct mapping from CUDA to HIP.
 */
__global__ void persistent_producer_consumer(
    volatile int* work_queue,
    volatile int* done_flags,
    float* data,
    int num_tasks,
    unsigned int spin_cycles)
{
    int tid = blockIdx.x * blockDim.x + threadIdx.x;

    while (true) {
        // Spin-poll for a task slot
        int task = -1;
        unsigned int start = __clock();

        while (task < 0) {
            // Try to atomically grab a task (0: available, 1: claimed)
            for (int i = 0; i < num_tasks; i++) {
                if (atomicCAS((int*)&done_flags[i], 0, 1) == 0) {
                    task = i;
                    break;
                }
            }
            
            // Bail if we've spun for too long. 
            // Unsigned subtraction handles wrap-around of the clock counter.
            if ((__clock() - start) > spin_cycles) {
                break;
            }
        }

        // If no task was found within the spin window, exit the persistent loop
        if (task < 0) break;

        // Perform the work
        // Note: Indexing assumes data is laid out as [task][thread_offset]
        data[task * blockDim.x + threadIdx.x] *= 1.5f;

        // Ensure the data write is visible to all threads before marking task as done
        __threadfence();
        
        // Mark complete (2: finished)
        done_flags[task] = 2; 
    }
}

// Example Host Wrapper for verification
void launch_kernel(float* d_data, int* d_flags, int num_tasks, int threads_per_block) {
    int blocks = 1; // Simplified for example
    unsigned int spin_limit = 1000000;

                       dim3(blocks), dim3(threads_per_block), 
                       0, 0, 
                       nullptr, // work_queue not used in original logic but passed
                       d_flags, 
                       d_data, 
                       num_tasks, 
                       spin_limit);
    
    hipDeviceSynchronize();
}

2.  **Memory Ordering:** The use of `volatile` combined with `__threadfence()` ensures that the "Done" flag (value 2) is never seen by another thread before the `data` multiplication is committed to global memory.
3.  **Complexity:** The time complexity per thread is $O(\text{tasks} \times \text{spin\_cycles})$, making this a "greedy" consumer model.

    // Normalized coordinates [0.0, 1.0]
    float u = (x + 0.5f) / (float)width;
    float v = (y + 0.5f) / (float)height;

    float val = hipTex2DLayered<float>(tex, u, v, layer);

    // Mix with neighbor layers
    float val_prev = (layer > 0)             ? hipTex2DLayered<float>(tex, u, v, layer - 1) : val;
    float val_next = (layer < num_layers - 1) ? hipTex2DLayered<float>(tex, u, v, layer + 1) : val;

    // Store result in linear output buffer
    output[layer * width * height + y * width + x] = 0.25f * val_prev + 0.5f * val + 0.25f * val_next;
}

#include <hip/hip_runtime.h>
#include <algorithm>

// Helper to simulate nanosleep on AMD hardware.
// Since AMD GPUs lack a direct __nanosleep equivalent, 
// we use a loop of NOPs/empty iterations to create a delay.
__device__ inline void hip_nanosleep(int cycles) {
    for (int i = 0; i < cycles; ++i) {
        // Use an assembly nop to prevent the compiler from optimizing the loop away
        asm volatile ("s_nop_save"); 
    }
}

__global__ void nanosleep_sync_kernel(
    volatile int* lock,
    float* shared_buffer,
    float* output,
    int N)
{
    int tid = blockIdx.x * blockDim.x + threadIdx.x;

    // Spin-lock with backoff (reduces memory traffic)
    if (threadIdx.x == 0) {
        int backoff = 8;
        // atomicCAS returns the old value. If old value was 0, we successfully set it to 1.
        while (atomicCAS(lock, 0, 1) != 0) {
            hip_nanosleep(backoff);  // AMD-compatible backoff
            backoff = (backoff < 256) ? backoff * 2 : 256; // Exponential backoff
        }
    }
    
    // Ensure all threads in the block wait for the lock-holder (thread 0)
    __syncthreads();

    // Critical section -- modify shared buffer
    if (tid < N) {
        shared_buffer[threadIdx.x] += 1.0f;
    }

    // Full device-level memory fence to ensure shared_buffer writes 
    // are visible to all other blocks before the lock is released.
    __threadfence();

    __syncthreads();
    if (threadIdx.x == 0) {
        // Release lock: atomically set lock back to 0
        atomicExch(lock, 0);
    }

    if (tid < N) {
        output[tid] = shared_buffer[threadIdx.x];
    }
}

// Example Host Launch Configuration
void launch_kernel(float* d_buffer, float* d_output, int* d_lock, int N) {
    int threadsPerBlock = 256;
    int blocksPerGrid = (N + threadsPerBlock - 1) / threadsPerBlock;

    hipLaunchKernelGGL(nanosleep_sync_kernel, 
                       dim3(blocksPerGrid), 
                       dim3(threadsPerBlock), 
                       0, 0, 
                       d_lock, d_buffer, d_output, N);
}
```

### Key Changes Summary:
1.  **`__nanosleep` $\rightarrow$ `hip_nanosleep`**: Replaced the NVIDIA-specific intrinsic with a `s_nop_save` assembly loop to prevent compiler optimization and provide a timing delay.
2.  **Type Casting**: Removed unnecessary `(int*)` casts on `lock` as the pointer is already defined as `volatile int*`.
3.  **`min()` $\rightarrow$ Ternary**: Replaced `min()` with a ternary operator to ensure compatibility across different HIP compiler versions without needing specific `<algorithm>` headers inside the device code.
4.  **Native HIP API**: Used `hipLaunchKernelGGL` for the launch sequence, adhering to ROCm standards.

/**
 * HIP Kernel for Peer-to-Peer Atomic Operations
 * 
 * Reasoning:
 * 1. atomicAdd(float*) is natively supported on AMD GCN/CDNA architectures.
 * 2. __threadfence_system() ensures visibility across the Infinity Fabric 
 *    and the PCIe bus to the CPU.
 * 3. The memory model assumes that the host has already called 
 *    hipDeviceEnablePeerAccess() or that the hardware is in a 
 *    coherent memory domain (e.g., MI200/MI300 series).
 */
__global__ void hip_peer_atomic_kernel(
    float* local_data,
    float* peer_data,     // Pointer into another GPU's memory via Infinity Fabric
    float* accumulator,   // Managed Memory buffer (HMM)
    int N)
{
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= N) return;

    float local_val = local_data[idx];

    // Cross-device atomic into peer GPU memory
    // On AMD, this utilizes the Infinity Fabric for remote memory access
    atomicAdd(&peer_data[idx], local_val);

    // System-level fence: Ensures the remote write is globally visible
    // before the subsequent operation on the managed accumulator.
    __threadfence_system();

    // Write into Managed Memory
    // This may trigger HMM page migration depending on the ROCm/Kernel version
    atomicAdd(&accumulator[idx % 1024], local_val * 0.5f);
}

// Example wrapper to demonstrate setup
void run_peer_atomic_example(int device0, int device1, int N) {
    float *d_local, *d_peer, *d_acc;

    hipSetDevice(device0);
    hipMalloc(&d_local, N * sizeof(float));
    
    hipSetDevice(device1);
    hipMalloc(&d_peer, N * sizeof(float));

    // Enable P2P access between devices
    hipSetDevice(device0);
    hipDeviceEnablePeerAccess(device1, 0);
    hipSetDevice(device1);
    hipDeviceEnablePeerAccess(device0, 0);

    // Allocate Managed Memory for the accumulator
    // hipMallocManaged is the equivalent to cudaMallocManaged
    hipMallocManaged(&d_acc, 1024 * sizeof(float));

    int blockSize = 256;
    int gridSize = (N + blockSize - 1) / blockSize;

    hipSetDevice(device0);
    hipLaunchKernelGGL(hip_peer_atomic_kernel, dim3(gridSize), dim3(blockSize), 0, 0, 
                       d_local, d_peer, d_acc, N);
    
    hipDeviceSynchronize();
}

1.  **`hipLaunchKernelGGL`**: Used instead of the triple-angle bracket syntax for maximum compatibility across different HIP compilers (clang/hipcc).
2.  **`hipMallocManaged`**: This is the direct replacement for `cudaMallocManaged`, enabling the "Unified Memory" behavior requested in the original code.
3.  **`hipDeviceEnablePeerAccess`**: Essential for the `peer_data` pointer to be valid. Without this, the kernel will trigger a memory fault (segmentation fault on GPU).
4.  **Complexity**: The time complexity remains $O(1)$ per thread, and space complexity is $O(N)$ for the data buffers.

// ROCm/HIP implementation of warp-specialized GEMM with double-buffered LDS
#pragma once
#include <hip/hip_runtime.h>
#include <hip/hip_fp16.h>
#include <rocwmma/rocwmma.h>

template <int BM, int BN, int BK, int WM, int WN, int WK>
struct GemmPolicy {
    static constexpr int kBlockM = BM, kBlockN = BN, kBlockK = BK;
    static constexpr int kWmmaM = WM, kWmmaN = WN, kWmmaK = WK;
    // AMD Wavefront size is 64
    static constexpr int kWarpsM = BM / WM;
    static constexpr int kWarpsN = BN / WN;
};

template <typename P>
struct SharedTile {
    __half A[P::kBlockM][P::kBlockK];
    __half B[P::kBlockK][P::kBlockN];
};

template <typename Policy>
__global__ void __launch_bounds__(256)
cutlass_gemm_kernel_hip(
    const __half* __restrict__ A,
    const __half* __restrict__ B,
    float*        __restrict__ C,
    int M, int N, int K)
{
    using P = Policy;
    __shared__ SharedTile<P> smem[2];

    // AMD Wavefront size is 64
    const int wave_id = threadIdx.x / 64; 
    const int wave_m  = wave_id / P::kWarpsN;
    const int wave_n  = wave_id % P::kWarpsN;

    // Initialize accumulator fragment
    rocwmma::fragment<rocwmma::accumulator, P::kWmmaM, P::kWmmaN, P::kWmmaK, float> acc_frag;
    rocwmma::fill_fragment(acc_frag, 0.0f);

    int smem_idx = 0;
    for (int k = 0; k < K; k += P::kBlockK) {
        // Load A into LDS
        for (int i = threadIdx.x; i < P::kBlockM * P::kBlockK; i += blockDim.x) {
            int row = i / P::kBlockK, col = i % P::kBlockK;
            int ga = (blockIdx.x * P::kBlockM + row) * K + (k + col);
            smem[smem_idx].A[row][col] = (blockIdx.x * P::kBlockM + row < M && k + col < K)
                ? A[ga] : __float2half(0.0f);
        }
        // Load B into LDS
        for (int i = threadIdx.x; i < P::kBlockK * P::kBlockN; i += blockDim.x) {
            int row = i / P::kBlockN, col = i % P::kBlockN;
            int gb = (k + row) * N + (blockIdx.y * P::kBlockN + col);
            smem[smem_idx].B[row][col] = (k + row < K && blockIdx.y * P::kBlockN + col < N)
                ? B[gb] : __float2half(0.0f);
        }
        
        __syncthreads();

        // Compute MMA
        for (int kk = 0; kk < P::kBlockK; kk += P::kWmmaK) {
            rocwmma::fragment<rocwmma::matrix_a, P::kWmmaM, P::kWmmaN, P::kWmmaK, __half, rocwmma::row_major> a_frag;
            rocwmma::fragment<rocwmma::matrix_b, P::kWmmaM, P::kWmmaN, P::kWmmaK, __half, rocwmma::col_major> b_frag;
            
            rocwmma::load_matrix_sync(a_frag, &smem[smem_idx].A[wave_m * P::kWmmaM][kk], P::kBlockK);
            rocwmma::load_matrix_sync(b_frag, &smem[smem_idx].B[kk][wave_n * P::kWmmaN], P::kBlockN);
            
            rocwmma::mma_sync(acc_frag, a_frag, b_frag, acc_frag);
        }
        
        smem_idx ^= 1;
        __syncthreads();
    }

    int global_m = blockIdx.x * P::kBlockM + wave_m * P::kWmmaM;
    int global_n = blockIdx.y * P::kBlockN + wave_n * P::kWmmaN;
    
    if (global_m < M && global_n < N) {
        rocwmma::store_matrix_sync(C + global_m * N + global_n, acc_frag, N, rocwmma::mem_row_major);
    }
}

using DefaultPolicy = GemmPolicy<64, 64, 16, 16, 16, 16>;

void launch_cutlass_gemm_hip(__half* A, __half* B, float* C, int M, int N, int K) {
    // Ensure blockDim.x is a multiple of 64 for AMD wavefronts
    dim3 grid((M + 63) / 64, (N + 63) / 64);
    int blockSize = 256; 
    hipLaunchKernelGGL(cutlass_gemm_kernel_hip<DefaultPolicy>, grid, blockSize, 0, 0, A, B, C, M, N, K);
}

// In ROCm, there is no direct 1:1 equivalent to cp.async.ca.shared.global.
// We implement this using standard vectorized loads. 
// The pipelining is managed via the buffer stages and __syncthreads().
template <int BYTES>
__device__ __forceinline__ void async_copy(void* dst_smem, const void* src_gmem) {
    if constexpr (BYTES == sizeof(half)) {
        *(half*)dst_smem = *(const half*)src_gmem;
    } else if constexpr (BYTES == 4) {
        *(int*)dst_smem = *(const int*)src_gmem;
    } else if constexpr (BYTES == 8) {
        *(long*)dst_smem = *(const long*)src_gmem;
    } else {
        // Fallback for arbitrary sizes
        memcpy(dst_smem, src_gmem, BYTES);
    }
}

    // No-op in HIP: synchronization is handled by __syncthreads()
}

#include <hip/half.h>
#include <hipwmma.h>
#include "pipeline_utils.h"

using namespace hipwmma;

constexpr int PIPE_STAGES = 3;
constexpr int TILE_M = 64, TILE_N = 64, TILE_K = 16;

struct alignas(128) SmemBuf {
    half A[PIPE_STAGES][TILE_M][TILE_K];
    half B[PIPE_STAGES][TILE_K][TILE_N];
};

__global__ void __launch_bounds__(256) pipeline_gemm(
    const half* __restrict__ A, const half* __restrict__ B,
    float* __restrict__ C, int M, int N, int K)
{
    // Shared memory allocation
    extern __shared__ uint8_t shared_mem[];
    SmemBuf* smem_buf = reinterpret_cast<SmemBuf*>(shared_mem);

    const int warp_id = threadIdx.x / 32;
    fragment<accumulator, 16, 16, 16, float> acc;
    fill_fragment(acc, 0.0f);

    int num_tiles = (K + TILE_K - 1) / TILE_K;

    // Prologue: prefill pipeline
    for (int s = 0; s < (PIPE_STAGES - 1 < num_tiles ? PIPE_STAGES - 1 : num_tiles); ++s) {
        int k_off = s * TILE_K;
        for (int i = threadIdx.x; i < TILE_M * TILE_K; i += blockDim.x)
            async_copy<sizeof(half)>(
                &smem_buf->A[s][i/TILE_K][i%TILE_K],
                A + (blockIdx.x * TILE_M + i/TILE_K) * K + k_off + i%TILE_K);
        for (int i = threadIdx.x; i < TILE_K * TILE_N; i += blockDim.x)
            async_copy<sizeof(half)>(
                &smem_buf->B[s][i/TILE_N][i%TILE_N],
                B + (k_off + i/TILE_N) * N + blockIdx.y * TILE_N + i%TILE_N);
        async_commit();
    }
    __syncthreads();

    // Steady-state: overlap compute with prefetch
    for (int tile = 0; tile < num_tiles; ++tile) {
        int prefetch = tile + PIPE_STAGES - 1;
        int cs = tile % PIPE_STAGES;

        // In HIP, we synchronize to ensure the data for the current stage 'cs' is ready
        __syncthreads();

        if (prefetch < num_tiles) {
            int k_off = prefetch * TILE_K, ps = prefetch % PIPE_STAGES;
            for (int i = threadIdx.x; i < TILE_M * TILE_K; i += blockDim.x)
                async_copy<sizeof(half)>(
                    &smem_buf->A[ps][i/TILE_K][i%TILE_K],
                    A + (blockIdx.x * TILE_M + i/TILE_K) * K + k_off + i%TILE_K);
            for (int i = threadIdx.x; i < TILE_K * TILE_N; i += blockDim.x)
                async_copy<sizeof(half)>(
                    &smem_buf->B[ps][i/TILE_N][i%TILE_N],
                    B + (k_off + i/TILE_N) * N + blockIdx.y * TILE_N + i%TILE_N);
            async_commit();
        }

        fragment<matrix_a, 16, 16, 16, half, row_major> a_frag;
        fragment<matrix_b, 16, 16, 16, half, col_major> b_frag;
        
        load_matrix_sync(a_frag, &smem_buf->A[cs][warp_id * 16][0], TILE_K);
        load_matrix_sync(b_frag, &smem_buf->B[cs][0][warp_id * 16], TILE_N);
        mma_sync(acc, a_frag, b_frag, acc);
    }

    __syncthreads();
    int out_m = blockIdx.x * TILE_M + warp_id * 16, out_n = blockIdx.y * TILE_N;
    if (out_m < M && out_n < N)
        store_matrix_sync(C + out_m * N + out_n, acc, N, mem_row_major);
}

#include <hip/hip_runtime.h>
#include <iostream>

// In ROCm, the "warp" equivalent is a "wavefront".
// Wavefront size is typically 64 on CDNA/GCN and 32 on RDNA.
// We use a constant or query it to ensure portability.
#ifndef WAVE_SIZE
#define WAVE_SIZE 64 
#endif

constexpr int PROD_WARPS = 2;
constexpr int CONS_WARPS = 6;
constexpr int TOT_WARPS = 8;
constexpr int SR = 64;
constexpr int SC = 64;

/**
 * ROCm/HIP Implementation of Warp-Specialized Stencil
 * 
 * Reasoning: 
 * 1. PTX 'bar.arrive/sync' are replaced with '__syncthreads()' because the 
 *    logic describes a hard dependency between the producer phase and consumer phase.
 * 2. 'threadIdx.x / 32' is updated to use WAVE_SIZE for AMD hardware compatibility.
 * 3. __launch_bounds__ is preserved as HIP supports this attribute.
 */
__global__ void __launch_bounds__(TOT_WARPS * WAVE_SIZE)
warp_specialized_stencil(const float* __restrict__ src, float* __restrict__ dst, int N)
{
    // Local Data Store (LDS) equivalent to CUDA __shared__
    __shared__ float smem[SR][SC];

    const int warp_id = threadIdx.x / WAVE_SIZE;
    const int lane = threadIdx.x % WAVE_SIZE;

    if (warp_id < PROD_WARPS) {
        // Producer Logic: Load data from global memory to LDS
        int rw = SR / PROD_WARPS;
        int rs = warp_id * rw;
        
        for (int r = rs; r < rs + rw; ++r) {
            for (int c = lane; c < SC; c += WAVE_SIZE) {
                int gi = blockIdx.x * SR * SC + r * SC + c;
                smem[r][c] = (gi < N) ? src[gi] : 0.0f;
            }
        }
        // Signal that producers have finished loading
        __syncthreads(); 
    } else {
        // Consumer Logic: Wait for producers to finish
        __syncthreads(); 
        
        int cid = warp_id - PROD_WARPS;
        int rw = SR / CONS_WARPS;
        int rs = cid * rw;
        
        for (int r = rs + 1; r < rs + rw - 1; ++r) {
            for (int c = lane + 1; c < SC - 1; c += WAVE_SIZE) {
                // 4-point stencil computation
                float v = 0.25f * (smem[r-1][c] + smem[r+1][c] +
                                   smem[r][c-1] + smem[r][c+1]);
                
                int go = blockIdx.x * SR * SC + r * SC + c;
                if (go < N) {
                    dst[go] = v;
                }
            }
        }
    }
}

// Example host wrapper for verification
void run_stencil(int N) {
    float *d_src, *d_dst;
    hipMalloc(&d_src, N * sizeof(float));
    hipMalloc(&d_dst, N * sizeof(float));

    dim3 grid(N / (SR * SC) + 1);
    dim3 block(TOT_WARPS * WAVE_SIZE);

    hipLaunchKernelGGL(warp_specialized_stencil, grid, block, 0, 0, d_src, d_dst, N);
    
    hipDeviceSynchronize();
    hipFree(d_src);
    hipFree(d_dst);
}

        max_val = fmaxf(max_val, scores[i]);
    max_val = warp_reduce_max(max_val);
    if (threadIdx.x == 0) smem_scratch[0] = max_val;
    __syncthreads();
    max_val = smem_scratch[0];

    float sum = 0.0f;
    for (int i = threadIdx.x; i < len; i += blockDim.x) {
        scores[i] = __expf(scores[i] - max_val);
        sum += scores[i];

    if (threadIdx.x == 0) smem_scratch[0] = sum;

    sum = smem_scratch[0];

    for (int i = threadIdx.x; i < len; i += blockDim.x)

        if (threadIdx.x < 16) { // Simplified mapping for 16x16 tile
            float local_score[16 * 16] = {0};
            for (int d = 0; d < DH; d += TILE_DH) {
                matmul_16x16x16_half_float(local_score, smem_q, smem_k + d, DH, DH);
            }
            for (int i = 0; i < 256; ++i) smem_scores[i] = local_score[i];
        }
        __syncthreads();

        for (int i = threadIdx.x; i < TILE_Q * TILE_K; i += blockDim.x) {
            int qi = i / TILE_K, ki = i % TILE_K;
            smem_scores[i] *= cfg.scale;
            if (cfg.causal_mask && (kv_start + ki > q_start + qi))
                smem_scores[i] = -1e9f;
        }
        __syncthreads();

        for (int qi = warp_id; qi < TILE_Q; qi += (blockDim.x / 64)) {
            float* row = smem_scores + qi * TILE_K;
            float new_max = running_max[qi];
            for (int ki = lane; ki < TILE_K; ki += 64)
                new_max = fmaxf(new_max, row[ki]);
            new_max = warp_reduce_max(new_max);

            float scale_old = expf(running_max[qi] - new_max);
            float new_sum = running_sum[qi] * scale_old;

            for (int ki = lane; ki < TILE_K; ki += 64) {
                row[ki] = expf(row[ki] - new_max);
                new_sum += row[ki];
            }
            new_sum = warp_reduce_sum(new_sum);

            for (int d = lane; d < DH; d += 64)
                smem_out[qi * DH + d] *= scale_old;

            if (lane == 0) {
                running_max[qi] = new_max;
                running_sum[qi] = new_sum;
            }
        }
        __syncthreads();

        for (int i = threadIdx.x; i < TILE_K * DH; i += blockDim.x) {
            int r = i / DH, c = i % DH;
            smem_v[r * DH + c] = (kv_start + r < S) ? V_bh[(kv_start + r) * DH + c] : __float2half(0.0f);
        }
        __syncthreads();

        // Score @ V Logic (Replacing WMMA)
        if (threadIdx.x < 16) {
            float local_out[16 * 16] = {0};
            for (int kk = 0; kk < TILE_K; kk += 16) {
                // Custom float-half matmul
                for (int i = 0; i < 16; ++i) {
                    for (int j = 0; j < 16; ++j) {
                        float sum = 0.0f;
                        for (int k = 0; k < 16; ++k) {
                            sum += smem_scores[i * 16 + k] * __half2float(smem_v[kk * DH + k * DH + j]);
                        }
                        local_out[i * 16 + j] += sum;
                    }
                }
            }
            for (int i = 0; i < 256; ++i) smem_out[i] += local_out[i];
        }
        __syncthreads();
    }

    for (int i = threadIdx.x; i < TILE_Q * DH; i += blockDim.x) {
        int qi = i / DH, d = i % DH;
        float norm = (running_sum[qi] > 0.0f) ? (1.0f / running_sum[qi]) : 0.0f;
        int out_row = q_start + qi;
        if (out_row < S)
            O_bh[out_row * DH + d] = smem_out[i] * norm;
    }
}

// FILE 4: main.hip

#include "attention.h"

void run_flash_attention(
    const half* d_Q, const half* d_K, const half* d_V,
    float* d_O,
    int B, int H, int S, int DH,
    bool causal)
{
    AttentionConfig cfg;
    cfg.batch_size = B; cfg.num_heads = H;
    cfg.seq_len = S;    cfg.head_dim = DH;
    cfg.scale = 1.0f / sqrtf((float)DH);
    cfg.causal_mask = causal;

    size_t smem = (TILE_Q + TILE_K + TILE_K) * DH * sizeof(half)
               + TILE_Q * TILE_K * sizeof(float)
               + TILE_Q * DH    * sizeof(float)
               + 64             * sizeof(float);

    dim3 grid((S + TILE_Q - 1) / TILE_Q, H, B);
    dim3 block(256);

    hipLaunchKernelGGL(flash_attention_kernel, grid, block, smem, 0, 0, d_Q, d_K, d_V, d_O, cfg);
    hipDeviceSynchronize();
}

 * for work using atomic operations and clock-based timeouts.
 * 
 * @param work_queue  Pointer to the work queue (not explicitly used in logic, but passed for structure)
 * @param done_flags  Pointer to flags indicating task status (0: available, 1: claimed, 2: done)

    volatile int* work_queue,
    volatile int* done_flags,

    int num_tasks,
    int spin_cycles)

    int tid = blockIdx.x * blockDim.x + threadIdx.x;

    while (true) {

        int task = -1;
        long long start = __clock64();

            // Try to atomically grab a task
            // atomicCAS returns the old value. If it was 0, we successfully claimed it.
            for (int i = 0; i < num_tasks; i++) {
                if (atomicCAS((int*)&done_flags[i], 0, 1) == 0) {
                    task = i;
                    break;
                }

            // This prevents threads from hanging if no work is available
            if ((__clock64() - start) > spin_cycles) break;

        // If no task was found within the timeout period, exit the kernel
        if (task < 0) break;

        // Perform the work: multiply data by 1.5
        // Coalesced memory access pattern
        data[task * blockDim.x + threadIdx.x] *= 1.5f;

        // Ensure all writes to data are visible to all threads before marking task as done
        __threadfence();

        // Mark the task as complete

/**
 * @brief A kernel demonstrating a spin-lock with nanosleep backoff.
 * 
 * This kernel uses a volatile lock to synchronize access to a shared buffer.
 * It employs __nanosleep to reduce memory bus contention during the spin phase,
 * which is a common optimization for high-contention locks on GPU architectures.
 * 
 * @param lock Pointer to a volatile integer used as a mutex.
 * @param shared_buffer Pointer to the shared data buffer.
 * @param output Pointer to the output buffer.
 * @param N Number of elements to process.
 */
__global__ void nanosleep_sync_kernel(
    volatile int* lock,
    float* shared_buffer,
    float* output,
    int N)
{
    int tid = blockIdx.x * blockDim.x + threadIdx.x;

    // Spin-lock with nanosleep backoff (reduces memory traffic)
    // Only the first thread in the block attempts to acquire the lock
    if (threadIdx.x == 0) {
        int backoff = 8;
        // atomicCAS returns the old value. If it was 0, we successfully set it to 1.
        while (atomicCAS((int*)lock, 0, 1) != 0) {
            // ROCm native intrinsic: sleep this thread N nanoseconds
            __nanosleep(backoff);  
            backoff = min(backoff * 2, 256);  // Exponential backoff to reduce contention
        }
    }
    
    // Ensure all threads in the block wait until the lock is acquired
    __syncthreads();

    // Critical section -- modify shared buffer
    // Only threads within the valid range N perform the operation
    if (tid < N) {
        shared_buffer[threadIdx.x] += 1.0f;
    }

    // Acquire/release fence to ensure all writes to shared_buffer are visible
    // to all threads in the device before the lock is released.
    __threadfence();  // Full device-level memory fence

    // Ensure all threads have completed their writes before releasing the lock
    __syncthreads();
    if (threadIdx.x == 0) {
        // Release lock by setting it back to 0
        atomicExch((int*)lock, 0);  
    }

    // Write the result to the output buffer
    if (tid < N) {
        output[tid] = shared_buffer[threadIdx.x];
    }
}

 * Reasoning:
 * 1. Memory Model: HIP supports peer-to-peer memory access via hipDeviceEnablePeerAccess().
 * 2. Atomics: atomicAdd on peer memory is supported on AMD GPU architectures.

 *    accessed by multiple GPUs with driver-managed migration.

    float* local_data,
    float* peer_data,     // Pointer into another GPU's memory via Infinity Fabric/PCIe
    float* accumulator,   // Unified Memory buffer
    int N)
{

    atomicAdd(&peer_data[idx], local_val);  // Peer atomic operation

// Warp-level max reduction using shuffle

    #pragma unroll

    val = warp_reduce_sum(val);
    if (lane == 0) smem[warp_id] = val;
    __syncthreads();
    val = (threadIdx.x < (blockDim.x / 32)) ? smem[lane] : 0.0f;
    if (warp_id == 0) val = warp_reduce_sum(val);
    return val;
}

// Attention config struct
struct AttentionConfig {
    int batch_size;
    int num_heads;
    int seq_len;
    int head_dim;
    float scale;          // 1 / sqrt(head_dim)
    bool causal_mask;

// Softmax over last dimension in shared memory
__device__ void smem_softmax(float* scores, int len, float* smem_scratch) {
    // Find max for numerical stability
    float max_val = -1e9f;
    for (int i = threadIdx.x; i < len; i += blockDim.x)
        max_val = fmaxf(max_val, scores[i]);
    max_val = warp_reduce_max(max_val);
    if (threadIdx.x == 0) smem_scratch[0] = max_val;
    __syncthreads();
    max_val = smem_scratch[0];

    // Exponentiate and sum
    float sum = 0.0f;
    for (int i = threadIdx.x; i < len; i += blockDim.x) {
        scores[i] = __expf(scores[i] - max_val);
        sum += scores[i];
    }
    sum = block_reduce_sum(sum, smem_scratch);
    if (threadIdx.x == 0) smem_scratch[0] = sum;
    __syncthreads();
    sum = smem_scratch[0];

    // Normalize
    for (int i = threadIdx.x; i < len; i += blockDim.x)
        scores[i] /= sum;
}

// FILE 3: attention.cu
// Full Flash-Attention style SDPA implementation with WMMA.
// ================================================================
#include "vec_math.h"
#include "attention.h"
#include <hip/hip_cooperative_groups.h>
namespace cg = cooperative_groups;
using namespace nvcuda;

// Tile sizes for the WMMA-based SDPA
constexpr int TILE_Q  = 16;   // Query tile rows
constexpr int TILE_K  = 16;   // Key tile cols
constexpr int TILE_DH = 16;   // Head-dim tile (must match WMMA_K)

// Flash-Attention SDPA kernel (single head, one block = one query tile)
__global__ void __launch_bounds__(256)
flash_attention_kernel(
    const __half* __restrict__ Q,   // [B, H, S, D]
    const __half* __restrict__ K,   // [B, H, S, D]
    const __half* __restrict__ V,   // [B, H, S, D]
    float*        __restrict__ O,   // [B, H, S, D]
    const AttentionConfig cfg)
{
    // One block handles one (batch, head, query_tile)
    const int batch_id = blockIdx.z;
    const int head_id  = blockIdx.y;
    const int q_tile   = blockIdx.x;

    const int S  = cfg.seq_len;
    const int DH = cfg.head_dim;
    const int stride_BH = cfg.num_heads * S * DH;
    const int stride_H  = S * DH;

    // Shared memory: Q-tile, K-tile, V-tile, attention scores, softmax scratch
    extern __shared__ float smem[];
    __half* smem_q      = (__half*)smem;                    // [TILE_Q][DH]
    __half* smem_k      = smem_q + TILE_Q * DH;            // [TILE_K][DH]
    __half* smem_v      = smem_k + TILE_K * DH;            // [TILE_K][DH]
    float*  smem_scores = (float*)(smem_v + TILE_K * DH);  // [TILE_Q][TILE_K]
    float*  smem_out    = smem_scores + TILE_Q * TILE_K;   // [TILE_Q][DH]
    float*  smem_tmp    = smem_out + TILE_Q * DH;          // temp for reductions

    const int warp_id = threadIdx.x / 32;
    const int lane    = threadIdx.x % 32;

    // Base pointers for this batch + head
    const __half* Q_bh = Q + batch_id * stride_BH + head_id * stride_H;
    const __half* K_bh = K + batch_id * stride_BH + head_id * stride_H;
    const __half* V_bh = V + batch_id * stride_BH + head_id * stride_H;
    float*        O_bh = O + batch_id * stride_BH + head_id * stride_H;

    // Load Q-tile into shared memory
    int q_start = q_tile * TILE_Q;
    for (int i = threadIdx.x; i < TILE_Q * DH; i += blockDim.x) {
        int r = i / DH, c = i % DH;
        int g = (q_start + r) * DH + c;
        smem_q[r * DH + c] = (q_start + r < S) ? Q_bh[g] : __float2half(0.0f);
    }

    // Initialize output accumulator and running softmax stats
    for (int i = threadIdx.x; i < TILE_Q * DH; i += blockDim.x)
        smem_out[i] = 0.0f;
    __shared__ float running_max[TILE_Q];
    __shared__ float running_sum[TILE_Q];
    if (threadIdx.x < TILE_Q) {
        running_max[threadIdx.x] = -1e9f;
        running_sum[threadIdx.x] = 0.0f;
    }
    __syncthreads();

    // Iterate over key-value tiles (Flash-Attention outer loop)
    int kv_limit = cfg.causal_mask ? (q_tile + 1) * TILE_K : (S + TILE_K - 1) / TILE_K;
    for (int kv_tile = 0; kv_tile < kv_limit; kv_tile++) {
        int kv_start = kv_tile * TILE_K;

        // Load K-tile
        for (int i = threadIdx.x; i < TILE_K * DH; i += blockDim.x) {
            int r = i / DH, c = i % DH;
            int g = (kv_start + r) * DH + c;
            smem_k[r * DH + c] = (kv_start + r < S) ? K_bh[g] : __float2half(0.0f);
        }
        __syncthreads();

        // QK^T via WMMA: score[q_row][k_col] = Q[q_row] . K[k_col]
        if (warp_id < (TILE_Q / 16)) {
            wmma::fragment<wmma::matrix_a, 16, 16, 16, __half, wmma::row_major> qa;
            wmma::fragment<wmma::matrix_b, 16, 16, 16, __half, wmma::col_major> kb;
            wmma::fragment<wmma::accumulator, 16, 16, 16, float> score_acc;
            wmma::fill_fragment(score_acc, 0.0f);
            for (int d = 0; d < DH; d += TILE_DH) {
                wmma::load_matrix_sync(qa, smem_q + warp_id * 16 * DH + d, DH);
                wmma::load_matrix_sync(kb, smem_k + d, DH);
                wmma::mma_sync(score_acc, qa, kb, score_acc);
            }
            wmma::store_matrix_sync(smem_scores + warp_id * 16 * TILE_K, score_acc,
                                    TILE_K, wmma::mem_row_major);
        }
        __syncthreads();

        // Scale scores and apply causal mask
        for (int i = threadIdx.x; i < TILE_Q * TILE_K; i += blockDim.x) {
            int qi = i / TILE_K, ki = i % TILE_K;
            smem_scores[i] *= cfg.scale;
            if (cfg.causal_mask && (kv_start + ki > q_start + qi))
                smem_scores[i] = -1e9f;  // Mask future positions
        }
        __syncthreads();

        // Online softmax (numerically stable, per Q-row)
        for (int qi = warp_id; qi < TILE_Q; qi += (blockDim.x / 32)) {
            float* row = smem_scores + qi * TILE_K;
            float new_max = running_max[qi];
            for (int ki = lane; ki < TILE_K; ki += 32)
                new_max = fmaxf(new_max, row[ki]);
            new_max = warp_reduce_max(new_max);

            float scale_old = __expf(running_max[qi] - new_max);
            float new_sum = running_sum[qi] * scale_old;

            for (int ki = lane; ki < TILE_K; ki += 32) {
                row[ki] = __expf(row[ki] - new_max);
                new_sum += row[ki];
            }
            new_sum = warp_reduce_sum(new_sum);

            // Rescale existing output accumulator
            for (int d = lane; d < DH; d += 32)
                smem_out[qi * DH + d] *= scale_old;

            if (lane == 0) {
                running_max[qi] = new_max;
                running_sum[qi] = new_sum;
            }
        }
        __syncthreads();

        // Load V-tile and accumulate: out += softmax(scores) @ V
        for (int i = threadIdx.x; i < TILE_K * DH; i += blockDim.x) {
            int r = i / DH, c = i % DH;
            int g = (kv_start + r) * DH + c;
            smem_v[r * DH + c] = (kv_start + r < S) ? V_bh[g] : __float2half(0.0f);
        }
        __syncthreads();

        // Score @ V via WMMA
        if (warp_id < (TILE_Q / 16)) {
            wmma::fragment<wmma::matrix_a, 16, 16, 16, float, wmma::row_major> sa;
            wmma::fragment<wmma::matrix_b, 16, 16, 16, __half, wmma::row_major> vb;
            wmma::fragment<wmma::accumulator, 16, 16, 16, float> out_acc;
            wmma::load_matrix_sync(out_acc, smem_out + warp_id * 16 * DH, DH,
                                   wmma::mem_row_major);
            wmma::load_matrix_sync(sa, smem_scores + warp_id * 16 * TILE_K, TILE_K);
            for (int kk = 0; kk < TILE_K; kk += 16) {
                wmma::load_matrix_sync(vb, smem_v + kk * DH, DH);
                wmma::mma_sync(out_acc, sa, vb, out_acc);
            }
            wmma::store_matrix_sync(smem_out + warp_id * 16 * DH, out_acc, DH,
                                    wmma::mem_row_major);
        }
        __syncthreads();
    }

    // Normalize output by running sum, then write to global memory
    for (int i = threadIdx.x; i < TILE_Q * DH; i += blockDim.x) {
        int qi = i / DH, d = i % DH;
        float norm = (running_sum[qi] > 0.0f) ? (1.0f / running_sum[qi]) : 0.0f;
        int out_row = q_start + qi;
        if (out_row < S)
            O_bh[out_row * DH + d] = smem_out[i] * norm;
    }
}

// ================================================================
// FILE 4: main.cu
// Host-side launcher and memory management.
// ================================================================
#include "attention.h"
#include <hip/hip_runtime.h>

void run_flash_attention(
    const __half* d_Q, const __half* d_K, const __half* d_V,
    float* d_O,
    int B, int H, int S, int DH,
    bool causal)
{
    AttentionConfig cfg;
    cfg.batch_size = B; cfg.num_heads = H;
    cfg.seq_len = S;    cfg.head_dim = DH;
    cfg.scale = 1.0f / sqrtf((float)DH);
    cfg.causal_mask = causal;

    // Shared memory: Q + K + V tiles + scores + output + tmp
    size_t smem = (TILE_Q + TILE_K + TILE_K) * DH * sizeof(__half)
               + TILE_Q * TILE_K * sizeof(float)
               + TILE_Q * DH    * sizeof(float)
               + 32             * sizeof(float);

    dim3 grid((S + TILE_Q - 1) / TILE_Q, H, B);
    dim3 block(256);

    flash_attention_kernel<<<grid, block, smem>>>(d_Q, d_K, d_V, d_O, cfg);
    hipDeviceSynchronize();
}
